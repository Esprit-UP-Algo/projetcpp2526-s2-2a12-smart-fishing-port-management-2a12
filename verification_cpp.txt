Points clés du flux de validation:

1. CREATE (onCreerUser):
   - Dialogue récupère les données
   - EmployeDAO::validateEmploye() valide TOUS les champs
   - Si erreur: affiche QMessageBox et arrête l'opération
   - Si succès: enregistre en base de données

2. UPDATE (onModifierUser):
   - Même flux de validation
   - EmployeeService::validateEmployee() valide avant modification
   - Erreur de validation: émission du signal "operationFailed"

3. DELETE (deleteEmployee):
   - Validation spécifique du CIN
   - EmployeDAO::validateCIN() vérifie l'identifiant
   - Suppression refusée si CIN invalide

QString EmployeDAO::validateCIN(const QString &cin)
{
    if (cin.isEmpty()) {
        return "Le CIN est obligatoire.";
    }
    
    if (!isAllDigits(cin)) {
        return "Le CIN doit contenir uniquement des chiffres.";
    }
    
    if (cin.length() != 8) {
        return "Le CIN doit contenir exactement 8 chiffres.";
    }
    
    return "";  // ✓ Valide
}

bool EmployeeService::matchesFilters(const EmployeUser &employee) const
{
    // Filtrer par nom/prénom
    if (!m_filterNom.isEmpty()) {
        bool nomMatch = employee.nom.contains(m_filterNom, Qt::CaseInsensitive);
        bool prenomMatch = employee.prenom.contains(m_filterNom, Qt::CaseInsensitive);
        if (!nomMatch && !prenomMatch)
            return false;
    }
    
    // Filtrer par email
    if (!m_filterEmail.isEmpty()) {
        if (!employee.email.contains(m_filterEmail, Qt::CaseInsensitive))
            return false;
    }
    
    // Filtrer par rôle
    if (m_filterRole != "Tous les rôles" && employee.role != m_filterRole)
        return false;
    
    // Filtrer par statut
    if (m_filterStatut != "Tous les statuts" && employee.statut != m_filterStatut)
        return false;
    
    return true;  // ✓ Tous les critères satisfaits
}

void EmployeeService::sortEmployees(QVector<EmployeUser> &employees) const
{
    std::sort(employees.begin(), employees.end(),
        [this](const EmployeUser &a, const EmployeUser &b) {
        
        int cmp = 0;
        
        switch(m_sortColumn) {
        case 0: // CIN
            cmp = a.cin.compare(b.cin, Qt::CaseInsensitive);
            break;
        case 1: // Rôle
            cmp = a.role.compare(b.role, Qt::CaseInsensitive);
            break;
        case 2: // Date de création
            cmp = (a.dateCreation < b.dateCreation) ? -1 
                : (a.dateCreation > b.dateCreation) ? 1 : 0;
            break;
        }
        
        return m_sortAscending ? (cmp < 0) : (cmp > 0);
    });
}

QVector<EmployeUser> EmployeMainWindow::filteredAndSortedUsers() const
{
    QVector<EmployeUser> result;
    
    // PARCOURS du vecteur + FILTRAGE en mémoire
    for (const auto &u : m_users)
        if (matchesUserSearch(u))
            result.append(u);
    
    // RÉCUPÉRATION des critères de tri de l'UI
    int sortCol  = ui->comboTriUser->currentIndex();   // 0=CIN, 1=Rôle, 2=Date
    bool ascending = ui->comboOrdreUser->currentIndex() == 0;
    
    // TRI EN MÉMOIRE
    std::sort(result.begin(), result.end(),
              [sortCol, ascending](const EmployeUser &a, const EmployeUser &b) {
        int cmp = 0;
        switch(sortCol) {
        case 0: cmp = a.cin.compare(b.cin, Qt::CaseInsensitive); break;
        case 1: cmp = a.role.compare(b.role, Qt::CaseInsensitive); break;
        case 2: cmp = (a.dateCreation < b.dateCreation) ? -1 
                     : (a.dateCreation > b.dateCreation ? 1 : 0); break;
        }
        return ascending ? cmp < 0 : cmp > 0;
    });
    
    return result;
}
